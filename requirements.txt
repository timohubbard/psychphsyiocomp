# Server-side engine dependencies. Python 3.11+ (verified on 3.14.7).
#
# Install:  py -m venv .venv && .venv/Scripts/python.exe -m pip install -r requirements.txt
# Models:   .venv/Scripts/python.exe tools/fetch_models.py

mediapipe>=1.0.1        # Apache-2.0 library. .task model bundle licence NOT verified — see README.
pyarrow>=25.0.0         # Parquet writer for the canonical tables
pytest>=8.0             # test runner
tzdata                  # IANA timezone DB — Windows Python does not ship it, and
                        # pyarrow needs it to read back the tz-aware timestamps
                        # in the runs table. Required at runtime, not just tests.

# Pulled in transitively by mediapipe, pinned here for visibility:
#   opencv-contrib-python, numpy, matplotlib, absl-py, sounddevice

# --- M2: facial Action Units -------------------------------------------------
py-feat>=2.1.3          # MIT code. AU model weights' own licence NOT verified —
                        # the manifest marks commercial_use "unknown" and the
                        # licence tool buckets that as unverified, not permissive.
                        # NOTE: py-feat DEFAULTS identity_model='arcface', a face
                        # recognition model with a non-commercial InsightFace
                        # weight. The engine passes identity_model=None and
                        # refuses it in params; see check_no_identity_model.

# Upper bounds below are NOT correctness constraints. On the Windows machine this
# was built on, an Application Control policy blocks freshly-published unsigned
# .pyd files, and pandas 3.0.6 / scikit-learn 1.9.1 were both refused at import
# with "An Application Control policy has blocked this file". These older builds
# are permitted. On Linux, or once those versions age into the policy's
# reputation data, the bounds can be lifted.
pandas<3                # 2.3.3 verified
scikit-learn<1.8        # 1.7.2 verified

# py-feat can read video directly, but that path needs torchcodec + FFmpeg
# shared libraries. The engine decodes with OpenCV instead, which is already a
# dependency, and feeds frames as tensors.

# --- M3: voice ---------------------------------------------------------------
opensmile>=2.6.0        # RESTRICTED. Verbatim from the openSMILE licence: "It is
                        # not allowed to use the open-source version of openSMILE
                        # for any sort of commercial product." Academic use is
                        # permitted, which is this build. This is the first
                        # genuinely restricted engine, so
                        # `npm run licenses:commercial-gate` now exits non-zero —
                        # correct and expected, not a failure.

imageio-ffmpeg>=0.6.0   # Bundles a static ffmpeg, used to demux the audio track
                        # out of the MediaRecorder WebM. Preferred over a system
                        # ffmpeg on purpose: codec support and defaults vary
                        # between builds, and audio extraction is not somewhere
                        # to inherit that variance.

# --- Transcription (optional) ------------------------------------------------
# Only needed for studies offering the `media_plus_transcript` consent tier.
# Everything else in the pipeline runs without it, and `tools/fetch_models.py`
# skips the Whisper checkpoint with a note rather than failing when it is absent.
#
# MIT throughout: faster-whisper and CTranslate2 are SYSTRAN's, and the Whisper
# weights are OpenAI's under MIT per the openai/whisper repository — a separate
# artifact whose licence line was read there rather than independently verified.
#
# Pulls in ctranslate2, onnxruntime, av and tokenizers, all of which ship native
# extensions. On the Windows machine this was built on they import cleanly under
# the Application Control policy that refused pandas 3 and scikit-learn 1.9,
# which is not a guarantee for future releases — if one of these is ever blocked
# at import, that is the same policy and not a broken install.
faster-whisper>=1.2.1
